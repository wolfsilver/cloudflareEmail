📡 API ENDPOINTS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

SYSTEM:
  GET  /              Web UI homepage
  GET  /api           API status and endpoint list
  POST /api/init      Initialize database tables

EMAIL:
  POST /api/send                    Send email to recipients
  POST /api/send-to-list/:listId    Send to entire mailing list

MAILING LISTS:
  GET    /api/lists                 Get all mailing lists
  GET    /api/lists/:id             Get single list details
  POST   /api/lists                 Create new mailing list
  PUT    /api/lists/:id             Update list name/description
  DELETE /api/lists/:id             Delete mailing list
  POST   /api/lists/:id/emails      Add email to list
  DELETE /api/lists/:id/emails/:email  Remove email from list

FILES:
  POST   /api/files                 Upload file to R2
  GET    /api/files/:id/:filename   Download/view file
  DELETE /api/files/:id/:filename   Delete file

🗄️  DATABASE SCHEMA
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

mailing_lists:
  - id (TEXT PRIMARY KEY)
  - name (TEXT NOT NULL)
  - description (TEXT)
  - created_at (TEXT NOT NULL)
  - updated_at (TEXT NOT NULL)

mailing_list_emails:
  - id (INTEGER PRIMARY KEY AUTOINCREMENT)
  - list_id (TEXT NOT NULL, FOREIGN KEY)
  - email (TEXT NOT NULL)
  - name (TEXT)
  - UNIQUE(list_id, email)

  ✓ schema.sql (Database schema)
  ✓ .gitignore (Git exclusions)

Deployment Steps:
  1. npm install
  2. wrangler r2 bucket create email-attachments
  3. wrangler d1 create email_database
  4. Update wrangler.toml with IDs
  5. npm run deploy
  6. POST /api/init to initialize DB
